-- 负责处理战斗中的武器状态与子弹切换逻辑

local Config_Common = require("Config.Config_Common")
local InventoryManager = require("System.InventoryManager")
local CombatManager = require("System.CombatManager")
local TeamManager = require("System.TeamManager") 

local WeaponManager = {}

-- 现在变成单纯让每个角色的武器冷却随帧倒流的管理器
function WeaponManager.Update(_deltaTime)
	for i = 1, 4 do
		local character = TeamManager.Characters[i]
		if character then
			local weapon = character:GetActiveWeapon()
			-- 仅扣减当前持有武器的内置 CD
			if weapon and weapon.FireCooldown > 0 then
				weapon.FireCooldown = weapon.FireCooldown - _deltaTime
			end
		end
	end
end

function WeaponManager.Init()
	-- C#事件的强解耦不再依赖 CS.InputManager，而是依靠底层全局桥接函数
	print("[WeaponManager] 战斗控制管线已切换为高速 Push 推送模式")
end

-- C# 层意图判断完毕，传过来的开火请求事件
function WeaponManager.OnFireInputPush(_characterID, _isHeld, _wasPressedThisFrame)
	local activeCharacter = TeamManager.GetCharacter(_characterID)
	if not activeCharacter then
		return
	end

	local weapon = activeCharacter:GetActiveWeapon()
	if not weapon then
		return
	end
	if weapon.FireCooldown > 0 then
		return
	end -- 还在卡 CD，直接抛弃本次扣留事件

	local weaponSubType = weapon.Config.WeaponSubType
	local attemptToFire = false

	if weaponSubType == Config_Common.WeaponSubType.Ranged then
		if weapon.Config.IsAuto then
			attemptToFire = _isHeld
		else
			attemptToFire = _wasPressedThisFrame
		end
	elseif weaponSubType == Config_Common.WeaponSubType.Melee then
		attemptToFire = _isHeld
	end

	-- CD准备就绪且触发逻辑有效，打进实际消耗子弹环节
	if attemptToFire then
		if weaponSubType == Config_Common.WeaponSubType.Ranged and weapon.CurrentAmmoCount <= 0 then
            local reserveAmmo = InventoryManager.GetItemCount(weapon.CurrentAmmoType)
            if reserveAmmo > 0 then
                print("[WeaponManager] 弹匣为空且有备弹，截获开火指令，自动转入换弹流程！")
                -- 核心：在内部直接复用并呼叫换弹管道！
                WeaponManager.OnReloadInputPush(_characterID)
            else
                print("[WeaponManager] 咔嗒...没子弹了，背包也没备弹了！")
                -- 给予一个极短的“卡壳冷却”，防止每帧疯狂 print 刷爆控制台
                weapon.FireCooldown = 0.2
            end
            
            return -- 既然是空枪，立刻终止，不用再走真实开火消耗逻辑了
        end
		
    	local success, elementType = CombatManager.Fire(activeCharacter)
		if success then
        	if weaponSubType == Config_Common.WeaponSubType.Ranged then
            	weapon.FireCooldown = weapon.Config.FireRate
        	elseif weaponSubType == Config_Common.WeaponSubType.Melee then
            	weapon.FireCooldown = weapon.Config.AttackRate
        	end
    	end
	end
end

-- C# 传过来的换弹请求事件
function WeaponManager.OnReloadInputPush(_characterID)
	local activeCharacter = TeamManager.GetCharacter(_characterID)
	if not activeCharacter then
		return
	end

	local success = CombatManager.Reload(activeCharacter)
	if success then
    	local weapon = activeCharacter:GetActiveWeapon()
    	-- 设置换弹无敌霸体真空期（武器进入硬锁定倒数时间）
    	weapon.FireCooldown = weapon.Config.ReloadTime

    	if CS.TeamController.Instance then
        	local tpsCC = CS.TeamController.Instance.teamMembers[_characterID - 1] -- Lua转C# List 0索引起点
        	if tpsCC then
        		tpsCC:ForceReload()
        	end
    	end
	end
end

-- C# 传过来的切子弹类型请求事件
function WeaponManager.OnSwitchAmmoInputPush(_characterID)
	local activeCharacter = TeamManager.GetCharacter(_characterID)
	if not activeCharacter then
		return
	end

	local weapon = activeCharacter:GetActiveWeapon()
	if not weapon or weapon.Config.WeaponSubType ~= Config_Common.WeaponSubType.Ranged then
		return
	end

	local nextAmmoID = WeaponManager.GetNextAvailableAmmo(weapon)
	if nextAmmoID and nextAmmoID ~= weapon.CurrentAmmoType then
		local success = CombatManager.SwitchAmmoType(activeCharacter, nextAmmoID)
		if success then
			weapon.FireCooldown = weapon.Config.ReloadTime
        	if CS.TeamController.Instance then
            	local tpsCC = CS.TeamController.Instance.teamMembers[_characterID - 1]
            	if tpsCC then
            		tpsCC:ForceReload()
            	end
        	end
		end
	else
		print("[WeaponManager] 背包中没有其他适用的元素子弹。")
	end
end

-- 在背包中寻找下一种可用的子弹
function WeaponManager.GetNextAvailableAmmo(_weapon)
	local currentAmmoID = _weapon.CurrentAmmoType
	local weaponClass = _weapon.Config.WeaponClass

	-- 根据配置表规则计算该枪械对应的基础子弹 ID
    -- 公式: AmmoIDStart(3000) + (WeaponClass * 100)
    local baseAmmoID = Config_Common.AmmoIDStart + (weaponClass * 100)

    -- 计算当前子弹的偏移量 (0=物理, 1=火, 2=雷, 3=冰)
    local currentOffset = currentAmmoID % 100

    -- 循环向后查找元素
    local maxElements = 4
    for i = 1, maxElements do
    	local nextOffset = (currentOffset + i) % maxElements
    	local checkAmmoID = baseAmmoID + nextOffset
    	-- 如果算出的这个 ID 在背包里有库存
    	if checkAmmoID ~= currentAmmoID and InventoryManager.GetItemCount(checkAmmoID) > 0 then
    		return checkAmmoID
    	end
    end

	-- 找了一圈都没有，返回原本的子弹
    return currentAmmoID
end
